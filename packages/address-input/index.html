import React, { useEffect, useRef } from 'react';
import '@abcubed3/nipost-address-input';

export function CheckoutAddressForm() {
  const addressRef = useRef(null);

  useEffect(() => {
    const el = addressRef.current;
    if (!el) return;

    const handleResolve = (e) => {
      const { postcode, address } = e.detail;
      console.log('Resolved Postcode:', postcode, address);
    };

    el.addEventListener('nipost-resolve', handleResolve);
    return () => el.removeEventListener('nipost-resolve', handleResolve);
  }, []);

  return (
    <div className="max-w-lg mx-auto">
      <nipost-address-input
        ref={addressRef}
        name="shipping_address"
        theme="light"
        required
      />
    </div>
  );
}
<template>
  <nipost-address-input
    name="shipping_address"
    @nipost-resolve="onAddressResolved"
    @nipost-change="onAddressChanged"
  />
</template>

<script setup>
import '@abcubed3/nipost-address-input';

function onAddressResolved(event) {
  console.log('Postcode:', event.detail.postcode);
  console.log('Full extraction:', event.detail.address);
}

function onAddressChanged(event) {
  console.log('Draft address:', event.detail.address);
}
</script>
